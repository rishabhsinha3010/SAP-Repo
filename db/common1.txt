namespace poapplication.common;

using { Country, Currency } from '@sap/cds/common';

type nameStr : String(50);
type id : Integer;

aspect address {
    drno     : String(20);
    street   : String(50);
    landmark : String(50);
    city     : String(100);
    postal   : Integer;
    state    : String(100);
    country  : Country;
    region   : String(20);
}

aspect fees {
    gross_fee : Decimal(10,2);
    currency : Currency;
    tax : Decimal(10,2);
    total_fee : Decimal(10,2);
}

type Gender : String(10) enum{
    M = 'Male';
    F = 'Female';
    U = 'Udisclosed';
}

type AmountT : Decimal(10,2) @(
    Semantics.amount.currencyCode : 'CURRENCY_CODE',
    sap.unit : 'CURRENCY_CODE'
);

type PhoneNumber : String(20) @assert.format : '^(?:(?:\+|00)?91[\-\s]?)?[6-9]\d{9}$';

type Email : String(40) @assert.format : '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}';

type Book : String(10) enum {
    S = 'Submitted';
    A = 'Approved';
    R = 'Rejected';
    I = 'Issued';
}
